FROM python:3.10-slim

# Set up a working directory
WORKDIR /code

# Copy requirements and install dependencies
COPY ./requirements.txt /code/requirements.txt
RUN pip install --no-cache-dir --upgrade -r /code/requirements.txt

# Create a cache directory for Hugging Face/SentenceTransformers 
# and grant access to Hugging Face's non-root user (UID 1000)
RUN mkdir -p /code/.cache && chmod -R 777 /code/.cache
ENV HF_HOME=/code/.cache
ENV TRANSFORMERS_CACHE=/code/.cache

# Copy the rest of your application code (including your backend folder)
COPY . .

# Expose Hugging Face's mandatory container port
EXPOSE 7860

# Run FastAPI pointing to your actual backend module on port 7860
CMD ["uvicorn", "backend.app:app", "--host", "0.0.0.0", "--port", "7860"]
